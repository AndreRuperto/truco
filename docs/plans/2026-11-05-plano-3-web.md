## Protocolo (lembrete — `apps/server/src/mesa/protocolo.ts`)

| Direção | Tipo | Payload |
|---|---|---|
| C → S | `acao` | `AcaoJogador` (o servidor sobrescreve `cadeira`) |
| C → S | `sinal` | `Sinal` |
| C → S | `fala` | `Fala` |
| C → S | `iniciar` | — (só o primeiro humano conectado; completa com bots) |
| C → S | `sincronizar` | — (reenvia o `estado` para quem pediu) |
| S → C | `estado` | `MsgEstado { minhaCadeira, cadeiras: InfoCadeira[], visao, prazo, minhasAcoes }` |
| S → C | `acao` | `{ cadeira, tipo }` — chega **antes** do `estado` resultante |
| S → C | `sinal` | `{ de, sinal, interceptado? }` |
| S → C | `fala` | `{ de, fala }` |
| S → C | `erro` | `{ mensagem }` |

Criação/entrada: `joinOrCreate('mesa', { variante, regras: { pontuacao, maoDe10, maoDe11, familia }, humanos, nivelBot, nome })` — o matchmaking filtra por `variante`, `humanos` e `nivelBot`; as `regras` valem as de quem criou a sala.

```ts
import type { Carta, Naipe } from './cartas';
import { forca, valorManilhaPaulista } from './forca';
import type { Variante } from './regras';

const MANILHAS_MINEIRO: Record<Naipe, Carta> = {
  paus: { valor: '4', naipe: 'paus' },
  copas: { valor: '7', naipe: 'copas' },
  espadas: { valor: 'A', naipe: 'espadas' },
  ouros: { valor: '7', naipe: 'ouros' },
};

/** `forca` de manilha começa em 11 (ver forca.ts). */
const PISO_MANILHA = 11;

/** A manilha de um naipe (zap = paus, copas, espadilha = espadas, pica-fumo = ouros). Paulista sem vira: null. */
export function manilhaDoNaipe(naipe: Naipe, variante: Variante, vira: Carta | null): Carta | null {
  if (variante === 'mineiro') return { ...MANILHAS_MINEIRO[naipe] };
  return vira ? { valor: valorManilhaPaulista(vira), naipe } : null;
}

export function ehManilha(c: Carta, variante: Variante, vira: Carta | null): boolean {
  if (variante === 'paulista' && !vira) return false;
  return forca(c, variante, vira) >= PISO_MANILHA;
}

/** A manilha mais forte da mão, ou null se não houver. */
export function melhorManilha(mao: readonly Carta[], variante: Variante, vira: Carta | null): Carta | null {
  let melhor: Carta | null = null;
  let forcaMelhor = 0;
  for (const c of mao) {
    if (!ehManilha(c, variante, vira)) continue;
    const f = forca(c, variante, vira);
    if (f > forcaMelhor) {
      melhor = c;
      forcaMelhor = f;
    }
  }
  return melhor;
}
```

```ts
export function amostrarMundo(v: Visao, rand: () => number, dica: DicaAmostra = {}): EstadoPartida {
  const m = v.mao;
  if (!m) throw new Error('não há mão em andamento');

  const conhecidas = new Set<string>();
  const marcar = (c: Carta | null) => {
    if (c) conhecidas.add(idCarta(c));
  };
  m.minhaMao.forEach(marcar);
  m.maoParceiro?.forEach(marcar);
  marcar(m.vira);
  m.mesa.forEach((j) => marcar(j.carta));
  m.rodadasJogadas.flat().forEach((j) => marcar(j.carta));

  const monte = embaralhar(
    novoBaralho().filter((c) => !conhecidas.has(idCarta(c))),
    rand,
  );
  const tirar = (): Carta => {
    const c = monte.pop();
    if (!c) throw new Error('monte esgotado ao amostrar');
    return c;
  };

  const parceiro = parceiroDe(v.cadeira);
  // O monte só tem cartas desconhecidas: se a carta sinalizada não está nele, a dica é impossível e é ignorada.
  let reservada: Carta | null = null;
  const alvo = dica.parceiroTem;
  if (alvo && !m.maoParceiro && m.qtdCartas[parceiro] > 0) {
    const i = monte.findIndex((x) => idCarta(x) === idCarta(alvo));
    if (i >= 0) reservada = monte.splice(i, 1)[0];
  }

  const maos: Carta[][] = ([0, 1, 2, 3] as Cadeira[]).map((c) => {
    if (c === v.cadeira) return m.minhaMao.map((x) => x ?? tirar());
    if (c === parceiro && m.maoParceiro) return m.maoParceiro.map((x) => ({ ...x }));
    if (c === parceiro && reservada) return [reservada, ...Array.from({ length: m.qtdCartas[c] - 1 }, tirar)];
    return Array.from({ length: m.qtdCartas[c] }, tirar);
  });
```

```ts
import {
  type Cadeira,
  type Carta,
  duplaDe,
  type EstadoPartida,
  manilhaDoNaipe,
  melhorManilha,
  type Naipe,
  type ResultadoRodada,
  type Variante,
} from '@truco/engine';
import type { Fala, Sinal } from './protocolo';

const SINAL_DO_NAIPE: Record<Naipe, Sinal> = { paus: 'zap', copas: 'copas', espadas: 'espadilha', ouros: 'picafumo' };
const NAIPE_DO_SINAL: Partial<Record<Sinal, Naipe>> = { zap: 'paus', copas: 'copas', espadilha: 'espadas', picafumo: 'ouros' };
const SEM_MANILHA: [Carta['valor'], Sinal][] = [
  ['3', 'tres'],
  ['2', 'dois'],
  ['A', 'as'],
];

/** O sinal que um bot manda ao parceiro com esta mão: a melhor manilha; senão 3, 2, Ás; senão "nada". */
export function sinalDaMao(mao: readonly Carta[], variante: Variante, vira: Carta | null): Sinal {
  const m = melhorManilha(mao, variante, vira);
  if (m) return SINAL_DO_NAIPE[m.naipe];
  for (const [valor, sinal] of SEM_MANILHA) if (mao.some((c) => c.valor === valor)) return sinal;
  return 'nada';
}

/** A carta exata que um sinal anuncia (só sinais de manilha), ou null. */
export function cartaDoSinal(sinal: Sinal, variante: Variante, vira: Carta | null): Carta | null {
  const naipe = NAIPE_DO_SINAL[sinal];
  return naipe ? manilhaDoNaipe(naipe, variante, vira) : null;
}

/** Comentário de um bot ao fim de uma rodada. `sorte` ∈ [0, 1) escolhe entre "boa" e "bora". */
export function falaDaRodada(resultado: ResultadoRodada, cadeira: Cadeira, sorte: number): Fala | null {
  if (resultado === 'empate') return null;
  if (resultado !== duplaDe(cadeira)) return 'vixe';
  return sorte < 0.5 ? 'boa' : 'bora';
}

/** Resultado da rodada que a jogada `antes → depois` fechou (inclusive a que encerrou a mão), ou null. */
export function rodadaFechada(antes: EstadoPartida, depois: EstadoPartida): ResultadoRodada | null {
  if (!antes.mao) return null;
  const n = antes.mao.rodadas.length;
  const rodadas = depois.mao ? depois.mao.rodadas : (depois.ultimaMao?.rodadas ?? []);
  return rodadas.length > n ? rodadas[rodadas.length - 1] : null;
}
```

> Use `rodadaFechada` só depois de uma ação `jogar` (pedir família zera a mão e deixaria a `ultimaMao` antiga à vista).

```ts
    const s = this.estado;
    if (!s?.mao || s.mao.tipo !== 'normal') return;
    if (!(this.o.sinaisBot ?? this.o.nivelBot === 'dificil')) return;
    for (const c of CADEIRAS) {

      this.enviarSinal(c, sinalDaMao(s.mao.maos[c], s.regras.variante, s.mao.vira));

  /** Entrega o sinal ao parceiro (humano), sorteia interceptações e guarda a dica do parceiro. */
  private enviarSinal(c: Cadeira, sinal: Sinal): void {
    const s = this.estado;
    const parceiro = parceiroDe(c);
    this.dicas[parceiro] = s?.mao ? cartaDoSinal(sinal, s.regras.variante, s.mao.vira) : null;
    const chance = this.o.chanceInterceptarSinal ?? 0;
    for (const alvo of CADEIRAS) {

      if (alvo === parceiro) this.o.enviar(this.ocupantes[alvo].id!, 'sinal', { de: c, sinal });
      else if (chance > 0 && this.rand() < chance) {
        this.o.enviar(this.ocupantes[alvo].id!, 'sinal', { de: c, sinal, interceptado: true });
      }
    }
  }

  /** Às vezes um bot comenta a rodada que acabou de fechar. */

    const chance = this.o.chanceFalaBot ?? 0;
    if (chance <= 0) return;
    const r = rodadaFechada(antes, depois);
    if (r === null || r === 'empate') return;
    const c = CADEIRAS.find((x) => !this.ocupantes[x].humano && this.rand() < chance);
    if (c === undefined) return;
    const fala = falaDaRodada(r, c, this.rand());
    if (fala && this.podeMandar(c)) this.paraHumanos((destino) => this.o.enviar(destino, 'fala', { de: c, fala }));
```

```ts
import type { Cadeira } from '@truco/engine';

export type Posicao = 'baixo' | 'direita' | 'cima' | 'esquerda';

const ORDEM: Posicao[] = ['baixo', 'direita', 'cima', 'esquerda'];

/**
 * Onde `outra` aparece na tela de quem senta em `minha`.
 * Convenção: eu embaixo; (minha + 1) % 4 — quem joga depois de mim — à direita
 * (sentido anti-horário); o parceiro em cima; (minha + 3) % 4 à esquerda.
 */
export function posicaoRelativa(minha: Cadeira, outra: Cadeira): Posicao {
  return ORDEM[(outra - minha + 4) % 4];
}

export function cadeiraNaPosicao(minha: Cadeira, pos: Posicao): Cadeira {
  return ((minha + ORDEM.indexOf(pos)) % 4) as Cadeira;
}
```

```ts
import { type Carta, ehManilha, type Naipe, type Valor, type Variante } from '@truco/engine';

export const SIMBOLO: Record<Naipe, string> = { ouros: '♦', espadas: '♠', copas: '♥', paus: '♣' };

const NOME_VALOR: Record<Valor, string> = {
  '4': '4',
  '5': '5',
  '6': '6',
  '7': '7',
  Q: 'Dama',
  J: 'Valete',
  K: 'Rei',
  A: 'Ás',
  '2': '2',
  '3': '3',
};

/** Nome da manilha pelo naipe — igual nas duas variantes (♣ > ♥ > ♠ > ♦). */
const NOME_MANILHA: Record<Naipe, string> = { paus: 'Zap', copas: 'Copas', espadas: 'Espadilha', ouros: 'Pica-fumo' };

export const ehVermelha = (c: Carta): boolean => c.naipe === 'ouros' || c.naipe === 'copas';

export const textoCarta = (c: Carta): string => `${c.valor}${SIMBOLO[c.naipe]}`;

export function nomeManilha(c: Carta, variante: Variante, vira: Carta | null): string | null {
  return ehManilha(c, variante, vira) ? NOME_MANILHA[c.naipe] : null;
}

/** Ex.: "Ás de espadas — Espadilha". Usado como nome acessível dos botões de carta. */
export function rotuloCarta(c: Carta, variante: Variante, vira: Carta | null): string {
  const base = `${NOME_VALOR[c.valor]} de ${c.naipe}`;
  const manilha = nomeManilha(c, variante, vira);
  return manilha ? `${base} — ${manilha}` : base;
}
```

```ts
  const p = v.mao?.trucoPendente;
  if (!p) return null;
  const quem = p.dupla === duplaDe(v.cadeira) ? 'Pedimos' : 'Pediram';
  return `${quem} ${nomePedido(p.proposto)}! Vale ${valorMao(v.regras, p.proposto)}.`;
}

export function placarRelativo(v: Visao): { nos: number; eles: number } {
  const d = duplaDe(v.cadeira);
  return { nos: v.placar[d], eles: v.placar[1 - d] };
}

export type Marcador = 'nos' | 'eles' | 'empate';

export function marcadoresRodadas(rodadas: ResultadoRodada[], minha: Cadeira): Marcador[] {
  const d = duplaDe(minha);
  return rodadas.map((r) => (r === 'empate' ? 'empate' : r === d ? 'nos' : 'eles'));
}

export interface ResumoMao {
  titulo: string;
  detalhe: string;
  /** true = minha dupla levou; null = ninguém. */
  nos: boolean | null;
}

export function resumoFimMao(u: VisaoResultadoMao, minha: Cadeira): ResumoMao {
  if (u.vencedor === null) return { titulo: 'Empatou!', detalhe: 'Ninguém pontua nesta mão', nos: null };
  const nos = u.vencedor === duplaDe(minha);
  const pontos = `+${u.pontos} ${u.pontos === 1 ? 'ponto' : 'pontos'}`;
  const motivo =
    u.motivo === 'rodadas'
      ? 'nas rodadas'
      : u.motivo === 'correu'
        ? nos
          ? 'eles correram'
          : 'nós corremos'
        : nos
          ? 'eles correram na mão de 10'
          : 'corremos na mão de 10';
  return { titulo: nos ? 'Nós levamos!' : 'Eles levaram!', detalhe: `${pontos} — ${motivo}`, nos };
}

export function resumoFimPartida(v: Visao): { titulo: string; placar: string; venceu: boolean } {
  const { nos, eles } = placarRelativo(v);
  const venceu = v.vencedor === duplaDe(v.cadeira);
  return { titulo: venceu ? 'Vitória!' : 'Derrota', placar: `Nós ${nos} × ${eles} Eles`, venceu };
}

export function textoFamilia(v: Visao, nomeDe: (c: Cadeira) => string): string | null {
  const f = v.ultimaFamilia;
  return f ? `${nomeDe(f.cadeira)} pediu família — cartas redistribuídas` : null;
}
```

```ts
  | { tipo: 'sinal'; de: Cadeira; sinal: Sinal; interceptado: boolean };

/** Evento efêmero do feed (balões, sinais). `em` = epoch ms local de chegada. */
export type Evento = NovoEvento & { id: number; em: number };

export const DURACAO_BALAO_MS = 2_500;

/**
 * Texto do balão para uma mensagem `acao`. O servidor manda `acao` ANTES do `estado`
 * resultante, então `anterior` é a visão de antes do pedido: truco propõe nivel + 1;
 * aumentar propõe o pendente + 1.
 */
export function textoDaAcao(a: MsgAcao, anterior: Visao | null): string | null {
  const m = anterior?.mao ?? null;
  switch (a.tipo) {
    case 'truco':
      return `${nomePedido((m?.nivel ?? 0) + 1).toUpperCase()}!`;
    case 'aumentar':
      return `${nomePedido((m?.trucoPendente?.proposto ?? 1) + 1).toUpperCase()}!`;
    case 'aceitar':
      return 'Aceito!';
    case 'correr':
      return 'Corro!';
    case 'familia':
      return 'Família!';
    default:
      return null;
  }
}

/** Balão atual de cada cadeira: o evento (ação/fala) mais recente dentro da duração. */
export function baloesAtivos(eventos: Evento[], agora: number, duracaoMs = DURACAO_BALAO_MS): Map<Cadeira, string> {
  const r = new Map<Cadeira, string>();
  for (const e of eventos) if (e.tipo !== 'sinal' && agora - e.em < duracaoMs) r.set(e.de, e.texto);
  return r;
}
```

```ts
import { nomePedido } from './textos';

export interface BotaoAcao {
  chave: 'aceitar' | 'aumentar' | 'truco' | 'familia' | 'mao10-sim' | 'correr' | 'mao10-nao';
  rotulo: string;
  /** Ação exatamente como veio em `minhasAcoes` — é o que se envia. */
  acao: AcaoJogador;
  tom: 'principal' | 'neutro' | 'perigo';
}

export interface AcoesUI {
  /** Índices de `minhaMao` jogáveis abertos. */
  jogaveis: number[];
  /** Índices jogáveis cobertos (2ª rodada em diante). */
  cobriveis: number[];
  botoes: BotaoAcao[];
}

const ORDEM: BotaoAcao['chave'][] = ['aceitar', 'aumentar', 'truco', 'familia', 'mao10-sim', 'correr', 'mao10-nao'];

/** Separa `minhasAcoes` em cartas jogáveis e botões, com rótulos ("Truco!", "Seis!"...). */
export function agruparAcoes(acoes: AcaoJogador[], m: VisaoMao | null): AcoesUI {
  const jogaveis: number[] = [];
  const cobriveis: number[] = [];
  const botoes: BotaoAcao[] = [];
  for (const a of acoes) {
    switch (a.tipo) {
      case 'jogar':
        (a.coberta ? cobriveis : jogaveis).push(a.indice);
        break;
      case 'truco':
        botoes.push({ chave: 'truco', rotulo: `${nomePedido((m?.nivel ?? 0) + 1)}!`, acao: a, tom: 'principal' });
        break;
      case 'aumentar':
        botoes.push({ chave: 'aumentar', rotulo: `${nomePedido((m?.trucoPendente?.proposto ?? 1) + 1)}!`, acao: a, tom: 'principal' });
        break;
      case 'aceitar':
        botoes.push({ chave: 'aceitar', rotulo: 'Aceitar', acao: a, tom: 'neutro' });
        break;
      case 'correr':
        botoes.push({ chave: 'correr', rotulo: 'Correr', acao: a, tom: 'perigo' });
        break;
      case 'familia':
        botoes.push({ chave: 'familia', rotulo: 'Família!', acao: a, tom: 'principal' });
        break;
      case 'mao10':
        botoes.push(
          a.jogar
            ? { chave: 'mao10-sim', rotulo: 'Jogar a mão', acao: a, tom: 'principal' }
            : { chave: 'mao10-nao', rotulo: 'Correr da mão', acao: a, tom: 'perigo' },
        );
        break;
    }
  }
  botoes.sort((x, y) => ORDEM.indexOf(x.chave) - ORDEM.indexOf(y.chave));
  return { jogaveis, cobriveis, botoes };
}
```

```ts
export interface AmbienteUrl {
  /** `import.meta.env.VITE_SERVIDOR`. */
  configurada?: string;
  /** `import.meta.env.DEV`. */
  dev: boolean;
  local: { protocol: string; hostname: string; host: string };
}

/**
 * Endereço do servidor Colyseus: `VITE_SERVIDOR` se definida; em dev, o mesmo host
 * da página na porta 2567 (funciona no celular pela rede local); em produção, a
 * mesma origem (o server vai servir o build — Plano 4).
 */
export function urlServidor(
  a: AmbienteUrl = { configurada: import.meta.env.VITE_SERVIDOR, dev: import.meta.env.DEV, local: window.location },
): string {
  if (a.configurada) return a.configurada;
  const ws = a.local.protocol === 'https:' ? 'wss:' : 'ws:';
  return a.dev ? `${ws}//${a.local.hostname}:2567` : `${ws}//${a.local.host}`;
```

```ts
import type { AcaoJogador } from '@truco/engine';
import type { Fala, MsgAcao, MsgErro, MsgEstado, MsgFala, MsgSinal, Sinal } from '@truco/protocolo';
import { type Evento, FALA_TEXTO, type NovoEvento, textoDaAcao } from '../logica/eventos';

/**
 * O mínimo de um `Room` do @colyseus/sdk 0.18 que a loja usa. O `Room` real satisfaz
 * isto (e os testes usam uma sala falsa).
 */
export interface SalaLike {
  readonly roomId: string;
  readonly reconnectionToken: string;
  onMessage(tipo: string, cb: (payload: any) => void): unknown;
  onLeave(cb: (code: number, reason?: string) => void): unknown;
  onDrop(cb: (code: number, reason?: string) => void): unknown;
  onReconnect(cb: () => void): unknown;
  send(tipo: string, payload?: unknown): void;
  leave(consented?: boolean): Promise<number>;
}

/** Todas as mensagens servidor → cliente; os handlers são registrados antes de `sincronizar`. */
export const TIPOS_MENSAGEM = ['estado', 'acao', 'sinal', 'fala', 'erro'] as const;

export type Status = 'fora' | 'conectando' | 'na-mesa' | 'reconectando';

export interface EstadoLoja {
  status: Status;
  /** Último `estado` do servidor (null antes do primeiro). */
  msg: MsgEstado | null;
  /** Feed efêmero (balões, sinais), do mais antigo ao mais novo. */
  eventos: Evento[];
  erro: { id: number; mensagem: string } | null;
}

export interface OpcoesLoja {
  agora?: () => number;
  /** Guarda/apaga o `reconnectionToken` (sessionStorage na app real). */
  persistirToken?: (token: string | null) => void;
  maxEventos?: number;
}

const INICIAL: EstadoLoja = { status: 'fora', msg: null, eventos: [], erro: null };

export function criarLoja(o: OpcoesLoja = {}) {
  const agora = o.agora ?? Date.now;
  const persistir = o.persistirToken ?? (() => {});
  const maxEventos = o.maxEventos ?? 30;

  let estado: EstadoLoja = INICIAL;
  let sala: SalaLike | null = null;
  let seq = 0;
  const ouvintes = new Set<() => void>();

  const mudar = (p: Partial<EstadoLoja>) => {
    estado = { ...estado, ...p };
    for (const f of ouvintes) f();
  };
  const evento = (e: NovoEvento) =>
    mudar({ eventos: [...estado.eventos, { ...e, id: ++seq, em: agora() }].slice(-maxEventos) });
  const erro = (mensagem: string) => mudar({ erro: { id: ++seq, mensagem } });
  const largar = () => {
    sala = null;
    persistir(null);
    mudar({ status: 'fora', msg: null, eventos: [] });
  };

  function conectar(s: SalaLike): void {
    sala = s;
    // Todos os handlers ANTES de qualquer envio: mensagem sem handler o SDK descarta.
    s.onMessage('estado', (m: MsgEstado) => {
      if (sala === s) mudar({ msg: m });
    });
    s.onMessage('acao', (a: MsgAcao) => {
      const texto = textoDaAcao(a, estado.msg?.visao ?? null);
      if (sala === s && texto) evento({ tipo: 'acao', de: a.cadeira, texto });
    });
    s.onMessage('fala', (f: MsgFala) => {
      if (sala === s) evento({ tipo: 'fala', de: f.de, texto: FALA_TEXTO[f.fala] ?? String(f.fala) });
    });
    s.onMessage('sinal', (x: MsgSinal) => {
      if (sala === s) evento({ tipo: 'sinal', de: x.de, sinal: x.sinal, interceptado: !!x.interceptado });
    });
    s.onMessage('erro', (e: MsgErro) => {
      if (sala === s) erro(e.mensagem);
    });
    s.onDrop(() => {
      if (sala === s) mudar({ status: 'reconectando' });
    });
    s.onReconnect(() => {
      if (sala !== s) return;
      persistir(s.reconnectionToken);
      mudar({ status: 'na-mesa' });
      s.send('sincronizar');
    });
    s.onLeave(() => {
      if (sala === s) largar();
    });
    persistir(s.reconnectionToken);
    mudar({ status: 'na-mesa', msg: null, eventos: [] });
    s.send('sincronizar');
  }

  return {
    subscribe(f: () => void): () => void {
      ouvintes.add(f);
      return () => {
        ouvintes.delete(f);
      };
    },
    getSnapshot: (): EstadoLoja => estado,
    conectar,
    marcarConectando: () => mudar({ status: 'conectando' }),
    /** Volta ao lobby; com mensagem, mostra o toast. */
    falhouConexao: (mensagem?: string) => {
      mudar({ status: 'fora' });
      if (mensagem) erro(mensagem);
    },
    enviarAcao: (a: AcaoJogador) => sala?.send('acao', a),
    enviarSinal: (x: Sinal) => sala?.send('sinal', x),
    enviarFala: (f: Fala) => sala?.send('fala', f),
    iniciar: () => sala?.send('iniciar'),
    /** Saída definitiva (o servidor põe um bot no lugar). */
    sair: async (): Promise<void> => {
      const s = sala;
      if (!s) return;
      try {
        await s.leave(true);
      } finally {
        if (sala === s) largar();
      }
    },
    limparErro: (id: number) => {
      if (estado.erro?.id === id) mudar({ erro: null });
    },
  };
}

export type Loja = ReturnType<typeof criarLoja>;
```

```ts
import type { OpcoesEntrada } from '../logica/lobby';
import type { Loja, SalaLike } from './loja';
import { urlServidor } from './url';

export const CHAVE_TOKEN = 'truco:reconexao';

/** O mínimo do `Client` do @colyseus/sdk 0.18 que usamos (permite testar sem rede). */
export interface ClienteLike {
  joinOrCreate(nomeSala: string, opcoes: object): Promise<unknown>;
  /** Recebe o `room.reconnectionToken` ("roomId:token"). */
  reconnect(token: string): Promise<unknown>;
}

export function lerToken(): string | null {
  try {
    return sessionStorage.getItem(CHAVE_TOKEN);
  } catch {
    return null;
  }
}

export function gravarToken(token: string | null): void {
  try {
    if (token) sessionStorage.setItem(CHAVE_TOKEN, token);
    else sessionStorage.removeItem(CHAVE_TOKEN);
  } catch {
    // storage indisponível: só perde a volta automática ao recarregar
  }
}

let padrao: Promise<ClienteLike> | null = null;

/** Carrega o SDK só quando precisa. O `Room` que ele devolve satisfaz `SalaLike`. */
function clientePadrao(): Promise<ClienteLike> {
  padrao ??= import('@colyseus/sdk').then(({ Client }) => new Client(urlServidor()) as unknown as ClienteLike);
  return padrao;
}

const mensagem = (e: unknown) => (e instanceof Error ? e.message : String(e));

export async function entrarNaMesa(loja: Loja, opcoes: OpcoesEntrada, cliente?: ClienteLike): Promise<void> {
  loja.marcarConectando();
  try {
    const c = cliente ?? (await clientePadrao());
    loja.conectar((await c.joinOrCreate('mesa', opcoes)) as SalaLike);
  } catch (e) {
    loja.falhouConexao(`Não foi possível entrar na mesa: ${mensagem(e)}`);
  }
}

let emAndamento: Promise<boolean> | null = null;

/**
 * Depois de recarregar a página: se há token, volta para a mesma cadeira (o servidor a
 * guarda por 60 s). Quedas de rede sem recarregar o SDK 0.18 já reconecta sozinho.
 */
export function tentarReconectar(loja: Loja, cliente?: ClienteLike): Promise<boolean> {
  const token = lerToken();
  if (!token) return Promise.resolve(false);
  emAndamento ??= (async () => {
    loja.marcarConectando();
    try {
      const c = cliente ?? (await clientePadrao());
      loja.conectar((await c.reconnect(token)) as SalaLike);
      return true;
    } catch {
      gravarToken(null);
      loja.falhouConexao();
      return false;
    } finally {
      emAndamento = null;
    }
  })();
  return emAndamento;
}
```

```ts
  const [agora, setAgora] = useState(() => Date.now());
```

```tsx
import type { Carta, Variante } from '@truco/engine';
import { ehVermelha, nomeManilha, rotuloCarta, SIMBOLO } from '../logica/cartas';
```

```tsx
import { type Carta, idCarta, type Variante } from '@truco/engine';
import { useState } from 'react';
import { rotuloCarta } from '../logica/cartas';
import { CartaVisual } from './CartaVisual';
import { cx } from './cx';

interface Props {
  /** Minha mão; null = carta virada (mão de ferro). */
  cartas: (Carta | null)[];
  jogaveis: number[];
  cobriveis: number[];
  variante: Variante;
  vira: Carta | null;
  aoJogar: (indice: number, coberta: boolean) => void;
}

export function Mao({ cartas, jogaveis, cobriveis, variante, vira, aoJogar }: Props) {
  const [cobrir, setCobrir] = useState(false);
  const podeCobrir = cobriveis.length > 0;
  const cobrindo = cobrir && podeCobrir;

  return (
    <div className="minha-mao" data-testid="minha-mao">
      <div className="cartas">
        {cartas.map((carta, i) => {
          const ok = (cobrindo ? cobriveis : jogaveis).includes(i);
          const nome = (carta ? rotuloCarta(carta, variante, vira) : `Carta virada ${i + 1}`) + (cobrindo ? ' (coberta)' : '');
          return (
            <button
              key={carta ? idCarta(carta) : `virada-${i}`}
              type="button"
              className={cx('carta-botao', ok && 'jogavel', cobrindo && 'cobrindo')}
              disabled={!ok}
              aria-label={nome}
              onClick={() => {
                aoJogar(i, cobrindo);
                setCobrir(false);
              }}
            >
              <CartaVisual carta={carta} variante={variante} vira={vira} />
            </button>
          );
        })}
      </div>
      {podeCobrir && (
        <button type="button" className="alternar-coberta" aria-pressed={cobrindo} onClick={() => setCobrir((x) => !x)}>
          Jogar coberta
        </button>
      )}
    </div>
  );
}
```

```ts
import type { Sinal } from '@truco/protocolo';

/** Emote e descrição de cada sinal (design: piscar = zap, sobrancelha = copas, boca torta = espadilha...). */
export const INFO_SINAL: Record<Sinal, { emoji: string; descricao: string }> = {
  zap: { emoji: '😉', descricao: 'Piscar — Zap' },
  copas: { emoji: '🤨', descricao: 'Sobrancelha — Copas' },
  espadilha: { emoji: '😏', descricao: 'Boca torta — Espadilha' },
  picafumo: { emoji: '😤', descricao: 'Bochecha — Pica-fumo' },
  tres: { emoji: '😛', descricao: 'Língua — tenho 3' },
  dois: { emoji: '😗', descricao: 'Biquinho — tenho 2' },
  as: { emoji: '🙄', descricao: 'Olhar pra cima — tenho Ás' },
  nada: { emoji: '😐', descricao: 'Cara fechada — nada' },
};

export interface SinalRecebido {
  sinal: Sinal;
  interceptado: boolean;
  nome: string;
}

export function textoSinalRecebido(s: SinalRecebido): string {
  const info = INFO_SINAL[s.sinal];
  const quem = s.interceptado ? `Interceptado de ${s.nome}` : `${s.nome} (parceiro)`;
  return `${quem}: ${info.emoji} ${info.descricao}`;
}
```

```tsx
import { useState } from 'react';
import { Assento } from '../componentes/Assento';

import { baloesAtivos } from '../logica/eventos';
import { cadeiraNaPosicao } from '../logica/posicoes';
import { prazoEhMeu } from '../logica/prazo';
import { type SinalRecebido, textoSinalRecebido } from '../logica/sinais';

  placarRelativo,
  resumoFimMao,
  resumoFimPartida,
  textoFamilia,
  textoPedidoPendente,
  valorMao,

import { useAgora } from '../rede/hooks';
import type { EstadoLoja, Loja } from '../rede/loja';

const DURACAO_SINAL_MS = 4_000;

interface Props {
  /** Precisa ter `msg.visao` (a App só mostra a mesa depois que a partida começa). */
  snap: EstadoLoja;
  acoes: Pick<Loja, 'enviarAcao' | 'enviarSinal' | 'enviarFala' | 'sair'>;
}

export function TelaMesa({ snap, acoes }: Props) {
  const agora = useAgora(250);
  const [painel, setPainel] = useState<'sinais' | 'falas' | null>(null);
  const msg = snap.msg!;
  const v = msg.visao!;
  const m = v.mao;
  const minha = msg.minhaCadeira;
  const variante = v.regras.variante;
  const vira = m?.vira ?? null;
  const ui = agruparAcoes(msg.minhasAcoes, m);
  const baloes = baloesAtivos(snap.eventos, agora);
  const { nos, eles } = placarRelativo(v);
  const nome = (c: Cadeira) => msg.cadeiras[c]?.nome ?? `Cadeira ${c + 1}`;
  const daVez = (c: Cadeira) => (msg.prazo ? msg.prazo.cadeiras.includes(c) : v.fase === 'jogando' && m?.vez === c);

  const ultimo = [...snap.eventos].reverse().find((e) => e.tipo === 'sinal');
  const sinal: (SinalRecebido & { em: number }) | null =
    ultimo && ultimo.tipo === 'sinal' ? { sinal: ultimo.sinal, interceptado: ultimo.interceptado, nome: nome(ultimo.de), em: ultimo.em } : null;

  const jogar = (indice: number, coberta: boolean) => {
    const a = msg.minhasAcoes.find((x) => x.tipo === 'jogar' && x.indice === indice && !!x.coberta === coberta);
    if (a) acoes.enviarAcao(a);
  };
  const alternar = (p: 'sinais' | 'falas') => setPainel((atual) => (atual === p ? null : p));

  const entreMaos = !m && v.fase === 'aguardandoDistribuicao';
  const familia = entreMaos ? textoFamilia(v, nome) : null;
  const resumoMao = entreMaos && !familia && v.ultimaMao ? resumoFimMao(v.ultimaMao, minha) : null;
  const fim = v.fase === 'fimPartida' ? resumoFimPartida(v) : null;

  return (
    <main className="tela-mesa">
      <Placar nos={nos} eles={eles} valor={m ? valorMao(v.regras, m.nivel) : null} />
      <div className="feltro">
        {(['cima', 'esquerda', 'direita'] as const).map((pos) => {
          const c = cadeiraNaPosicao(minha, pos);
          return (
            <Assento
              key={pos}
              posicao={pos}
              info={msg.cadeiras[c]}
              qtdCartas={m?.qtdCartas[c] ?? 0}
              daVez={daVez(c)}
              balao={baloes.get(c) ?? null}
              maoVisivel={c === parceiroDe(minha) ? (m?.maoParceiro ?? null) : null}
              variante={variante}
              vira={vira}
            />
          );
        })}
        <CentroMesa mao={m} minha={minha} variante={variante} />
        <Assento
          posicao="baixo"
          info={msg.cadeiras[minha]}
          qtdCartas={0}
          daVez={daVez(minha)}
          balao={baloes.get(minha) ?? null}
          maoVisivel={null}
          variante={variante}
          vira={vira}
        />
      </div>
      {prazoEhMeu(msg) && msg.prazo && <Cronometro prazo={msg.prazo} agora={agora} />}
      <BarraAcoes botoes={ui.botoes} pedido={textoPedidoPendente(v)} aoEscolher={acoes.enviarAcao} />
      {m && <Mao cartas={m.minhaMao} jogaveis={ui.jogaveis} cobriveis={ui.cobriveis} variante={variante} vira={vira} aoJogar={jogar} />}
      <nav className="barra-social" aria-label="Sinais e falas">
        <button type="button" aria-pressed={painel === 'sinais'} onClick={() => alternar('sinais')}>
          Sinais
        </button>
        <button type="button" aria-pressed={painel === 'falas'} onClick={() => alternar('falas')}>
          Falas
        </button>
      </nav>
      {painel === 'sinais' && (
        <PainelSinais
          ultimo={sinal}
          aoEnviar={(s) => {
            acoes.enviarSinal(s);
            setPainel(null);
          }}
        />
      )}
      {painel === 'falas' && (
        <PainelFalas
          aoEnviar={(f) => {
            acoes.enviarFala(f);
            setPainel(null);
          }}
        />
      )}
      {painel !== 'sinais' && sinal && agora - sinal.em < DURACAO_SINAL_MS && (
        <div className="sinal-flutuante" role="status">
          {textoSinalRecebido(sinal)}
        </div>
      )}
      {familia && <FimDeMao titulo="Família!" detalhe={familia} />}
      {resumoMao && <FimDeMao titulo={resumoMao.titulo} detalhe={resumoMao.detalhe} />}
      {fim && <FimDePartida titulo={fim.titulo} placar={fim.placar} venceu={fim.venceu} aoJogarDeNovo={() => void acoes.sair()} />}
    </main>
  );
}
```
